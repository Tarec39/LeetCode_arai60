```python3
class Solution:
    def addTwoNumbers(self, l1: ListNode | None, l2: ListNode | None) -> ListNode | None:
        if l1 is None and l2 is None:
            return None
                
        new_l = ListNode()
        node = new_l
        count = 0

        while l1 or l2 or count:
            a,b = 0,0
            if l1:
               a = l1.val
            if l2:
                b = l2.val
            
            total = a + b + count
            digit = total % 10
            count = total // 10

            node.next = ListNode(digit)
            node = node.next
            
            if l1:
                l1 = l1.next
            if l2:
                l2 = l2.next

        return new_l.next
```
