```python3
class Solution(object):
    def isValid(self, s):
        """
        :type s: str
        :rtype: bool
        """
        stack = []
        pairs = {
            "}" : "{",
            "]" : "[",
            ")" : "("
        }

        for bracket in s:
            if bracket not in pairs:
                stack.append(bracket)
                continue
            
            if pairs[bracket] != stack.pop():
                return False
        
        return True

```
