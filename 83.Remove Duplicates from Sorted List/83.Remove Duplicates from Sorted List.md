## 第一段階
#### 2ポインターの解法
```python3
class Solution:
    def deleteDuplicates(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None
        
        node = head
        while node is not None:
            if node.val = node.next.val:
                node.next = node.next.next    
            node = node.next
    
        return head
```
#### ハッシュテーブルとダミーノード
ひねり出した感じがある
```python3
class Solution:
    def deleteDuplicates(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None

        seen = set()
        
        node = head
        new_head = ListNode(None)
        new_node = new_head
        
        while node is not None:
            if node.val not in seen:
                new_node.next = node
                new_node = new_node.next
            
            seen.add(node.val)
            node = node.next

        new_node.next = None
        
        return new_head.next
```
