```python3
class Solution:
    def deleteDuplicates(self, head: ListNode | None) -> ListNode | None:
        if head is None:
            return None
        
        dummy = ListNode(0, head)
        prev = dummy
        node = head

        while node:
            if node.next and node.val == node.next.val:
                copy_val = node.val
                while copy_val == node.val:
                    node = node.next
                prev.next = node
            else:
                prev = node
                node = node.next
        
        return dummy.next
```
